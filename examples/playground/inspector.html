// Assets are processed by the build plugin.
import sculpture from "./orbit.glb?asset";

const scene = app.createScene();
const model = scene.addModel(sculpture);

model.animation.play("Idle");
model.on("click", () => {
  model.animation.play("Pulse");
});

app.present(scene);
